\section{Deferred Proofs}

\subsection{Proofs for Section~\ref{sec:functional-calm}}
\label{app:proofs-sec2}

\begin{proof}[Proof of Lemma~\ref{lem:run-residual}]
The run starts at the canonical state $\sigma_P(i_0)$, whose table is
$P[\mathrm{in}\InputGe i_0]$. Each input event $\mathsf{input}(j)$ keeps only
the rows with $\mathrm{in}\InputGe j$ (Equation~\ref{eq:input-event}), and each
output event $\mathsf{output}(o)$ keeps only the rows with
$\mathrm{out}\OutputGe o$ (Equation~\ref{eq:output-action}). The input
conditions reduce to the last one, $\mathrm{in}\InputGe i$, because each later
input is an extension of the earlier ones. One output condition remains for each
emitted output.
\end{proof}

\begin{proof}[Proof of Proposition~\ref{prop:confluence}]
A common state with current input $j$ has a row at $j$, and since rows are
never added, that row belongs to every $R_r$.  Conversely, let $(j,u)$ belong
to every $R_r$.  Then $\mathsf{input}(j)$ is defined at every $(i_r,R_r)$, and
$\mathsf{output}(u)$ is defined at $(j,R_r[\mathrm{in}\InputGe j])$, since
$u\in R_r(j)$.  By
Lemma~\ref{lem:run-residual}, $(j,u)\in R_r$ says that $u$ \outrel every output
emitted on run $r$.  After $\mathsf{input}(j)$ and $\mathsf{output}(u)$, the
same lemma leaves exactly
$P[\mathrm{in}\InputGe j,\ \mathrm{out}\OutputGe u]$, independent of the
starting run.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:confluence-legality}]
We use legality in a form that covers both functional and nondeterministic
specs: $o$ is legal at $(i,R)$ when, for every row $(k,x)\in R$, some
$u\in R(k)$ \outrel both $o$ and $x$.  In a functional table the only row at
$k$ is $(k,x)$, so this says $x$ \outrel $o$
(Definition~\ref{def:exposable}); with several outputs per input it says that
$\Excluded_o$ is empty (Section~\ref{sec:lifted-output-events}).

($\Rightarrow$) Suppose a run ends at $(i,R)$ and $o\in R(i)$ is not legal
there.  Then some $(k,x)\in R$ has no $u\in R(k)$ \outrelpart both $o$ and
$x$.  We extend the run in two ways: by $\mathsf{output}(o)$, and by
$\mathsf{input}(k)$ followed by $\mathsf{output}(x)$, which is defined since
$(k,x)$ remains after $\mathsf{input}(k)$.  The input $k$ is a common \indesc
of the two current inputs.  A row at $k$ in both resulting tables would be a
row $(k,u)\in R$ with $u$ \outrelpart both $o$ and $x$, and there is none.  By
Proposition~\ref{prop:confluence}, the two runs are not confluent at $k$.

($\Leftarrow$) Suppose every output that a run can emit is legal, and let runs
end at $(i_r,R_r)$, $r=1,\ldots,m$, with common \indesc $j$.  Each $R_r$ keeps a
row at its current input, say $(i_r,u_r)$, and by Lemma~\ref{lem:run-residual}
$u_r$ \outrel every output emitted on run $r$.  The canonical state
$\sigma_P(i_r)$ is reached by the empty run and has $u_r$ as an output, so $u_r$
is legal there.  By totality there is some $v_0\in P(j)$.  For $r=1,\ldots,m$
in turn, $(j,v_{r-1})$ is a row of $\sigma_P(i_r)$, since $j\InputGe i_r$, so
legality of $u_r$ gives $v_r\in P(j)$ \outrelpart both $u_r$ and $v_{r-1}$.
Then $v_m$ \outrel every $u_r$.  By Lemma~\ref{lem:run-residual},
$(j,v_m)\in R_r$ for every $r$, since $j\InputGe i_r$ and $v_m$ \outrel every
output emitted on run $r$.  Proposition~\ref{prop:confluence} gives
confluence at $j$.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:state-by-input}]
By Lemma~\ref{lem:run-residual}, the run's remainder table is
$P[\mathrm{in}\InputGe i,\ \mathrm{out}\OutputGe o_1,\ \ldots,\
\mathrm{out}\OutputGe o_m]$, where each $o_\ell=f(i_\ell)$ for some
$i_\ell\InputOrd i$.  Every row whose input \inrel $i$ has the form $(k,f(k))$ with
$i_\ell\InputOrd k$, so monotonicity gives $o_\ell\OutputOrd f(k)$, and the
output conditions remove nothing.
\end{proof}

\begin{proof}[Proof of Proposition~\ref{prop:functional-exposure-boundary}]
Suppose $f$ is monotone, and let a \commitment-free run reach $(i,R)$.  By Corollary~\ref{cor:state-by-input},
$(i,R)=\sigma_P(i)$, so the legal $j\InputGe i$ are exactly the \indescs of
$i$, and monotonicity gives $f(i)\OutputOrd f(j)$ at each, so $f(i)$ is a
legal output (Definition~\ref{def:exposable}).
Conversely, if every such output is a legal output, then at $\sigma_P(i)$,
which the empty run reaches, legality of $f(i)$ gives $f(i)\OutputOrd f(j)$
for every \indesc $j$ of $i$, so $f$ is monotone.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:functional-calm}]
By Lemma~\ref{lem:confluence-legality}, the runs are consistent exactly when
every output that a run can emit is legal.  This condition is equivalent to
legality at every state reachable without \commitment.  One direction is
immediate, since such states are reached by runs.  For the other, suppose every
output at every state reachable without \commitment is legal.  Then, by
induction along any run, each of its output events is legal, so the run is
\commitment-free and every state it reaches is reachable without \commitment;
hence every output it can emit is legal.  By
Proposition~\ref{prop:functional-exposure-boundary}, legality at every state
reachable without \commitment holds exactly when $f$ is monotone.
\end{proof}

\subsection{Proofs for Section~\ref{sec:complete-calm}}
\label{app:proofs-sec3}

\begin{proof}[Proof of Proposition~\ref{prop:common-expansion-form}]
Suppose the spec satisfies Lifted Monotonicity, and let $(i_1,o_1),\ldots,(i_k,o_k)$ be rows
with common \indesc $j$.  If $k=0$, any $u\in\fiber{j}$ will do, and since
$\Spec$ is total, $j$ appears in some row $(j,u)$.  Otherwise, since $i_r\InputOrd j$, Relational Monotonicity gives, for each
$r$, a $u_r\in\fiber{j}$ with $o_r\OutputOrd u_r$.  The $u_r$ all lie in the
one fiber $\fiber{j}$, which is directed, so some $u\in\fiber{j}$ \outrel
every $u_r$, hence every $o_r$.  Conversely, let $i\InputOrd j$ and
$X\subseteq\fiber{i}$ be nonempty and finite.  The rows $\{(i,x):x\in X\}$
have common \indesc $j$, so some $u\in\fiber{j}$ \outrel all of $X$; that is
$\fiber{i}\LiftedOrd\fiber{j}$.
\end{proof}

The Survivor test (Proposition~\ref{prop:confluence}) and
Lemma~\ref{lem:run-residual} combine into the form that the later arguments
use.

\begin{lemma}[Reconciliation]
\label{lem:reconciliation}
Let runs end at $(i_r,R_r)$, $r=1,\ldots,m$, let $j$ be a common \indesc of
the $i_r$, and let $X$ be the set of outputs emitted on any of them.  The runs
are confluent at $j$ if and only if some $u\in\fiber{j}$ \outrel every output
in $X$.
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:run-residual}, $(j,u)\in R_r$ exactly when $u\in\fiber{j}$
\outrel every output emitted on run $r$, and Proposition~\ref{prop:confluence}
gives the claim.
\end{proof}

\begin{proof}[Proof of Proposition~\ref{prop:exposure-boundary}]
By Proposition~\ref{prop:common-expansion-form} it suffices to treat the
common-extension condition.  Suppose it holds.  Let $(i,R)$ be reached by a run
(\commitment-free or not), and let $o\in R(i)$, $j\InputGe i$ legal, and
$x\in R(j)$.  The rows $(i,o)$ and $(j,x)$ of $P$ have common \indesc $j$, so
the condition supplies $u\in P(j)$ \outrelpart both $o$ and $x$.  By
Lemma~\ref{lem:run-residual}, $(j,u)\in R$, since $j\InputGe i$ and $u$
\outrel $o$, which \outrel every output emitted on the run.  So $u\in R(j)$
\outrel $o$ and $x$, and the legal-output test
(Proposition~\ref{prop:exposability}) makes $o$ a legal output.
Conversely, suppose every such output is a legal output, and let rows
$(i_1,o_1),\ldots,(i_k,o_k)$ have common \indesc $j$.  By totality there is some
$u_0\in P(j)$.  For $r=1,\ldots,k$ in turn, the legal-output test for $o_r$ at
$\sigma_P(i_r)$, applied with $x=u_{r-1}$, gives $u_r\in P(j)$ \outrelpart both
$o_r$ and $u_{r-1}$.  Then $u_k$ \outrel every $o_r$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:complete-calm}]
The proof of Lemma~\ref{lem:confluence-legality} covers nondeterministic specs,
so the runs are consistent exactly when every output that a run can emit is
legal.  As in the proof of Theorem~\ref{thm:functional-calm}, this holds exactly
when every output at every state reachable without \commitment is legal.  By
Proposition~\ref{prop:exposure-boundary}, that holds exactly when the spec
satisfies Lifted Monotonicity.
\end{proof}

\paragraph{Pairs suffice for the common-extension condition.}
\label{app:pairs-suffice}
Suppose every two rows with a common \indesc $j$ have a common \outnoun in
$\fiber{j}$.  Then the condition holds for every finite family of rows with
common \indesc $j$.  We replace the first two rows of the family by a row
$(j,u)$ whose output $u$ \outrel both, and repeat until one row remains; a
single row, paired with any row at $j$, has an \outnoun in $\fiber{j}$.  The
pairwise hypothesis must cover all rows of $P$, including rows at $j$ itself,
since the replacement rows are at $j$.  Checking only rows at single-proposal
inputs does not suffice, even with families of up to $k$ rows: in the $k$-set agreement example of
Appendix~\ref{app:additional-applications}, every family of at most $k$ rows
at such inputs is compatible at each common \indesc, but the $k+1$ proposal
rows are not.

\subsection{Proofs for Section~\ref{sec:distributed-execution}}
\label{app:proofs-sec4}

\begin{proof}[Proof of Proposition~\ref{prop:distributed-consistency}]
A cut is confluent at $j$ exactly when its participant runs are
(\S\ref{sec:distributed-consistency}).  The participant runs at any cut form a
finite family of runs, so consistent runs give consistent histories.
Conversely, every finite family of runs is the family of participant runs, at
the final cut, of a message-free history with one participant per run, so
consistent histories give consistent runs.
\end{proof}

\paragraph{Exchanging outputs.}
\label{app:exchanging-outputs}
Suppose participants share their \emph{outputs} in the data $d$ of messages
(\S\ref{sec:distributed-consistency}).  A participant that receives an
output $o$ its sender has emitted has two options, and may take both.  First,
if $o\in R(i)$ at its own current input $i$, it can emit $o$ itself.  That is
an ordinary output event, already covered by
Definition~\ref{def:distributed-history}.  Second, it can silently honor $o$
by restricting its table to $R[\mathrm{out}\OutputGe o]$.  The restricted
table may have no row at the participant's current input, so it is a table
but not always a specification state.  Silent honoring is therefore not a
specification event, and we reason about it on tables.

Silent honoring never changes whether a cut is confluent.  It cannot make a
cut confluent, because it only removes rows, and confluence asks for a row
that survives everywhere (Proposition~\ref{prop:confluence}).  It cannot make
a cut non-confluent either.  A row that survives at a common \indesc already
\outrel every output emitted in the cut (Lemma~\ref{lem:run-residual}).  That
includes $o$, which was emitted earlier, so the restriction keeps the row.
The confluence of a cut therefore depends on which outputs were emitted,
not on who has heard about them.

\begin{proof}[Proof of Proposition~\ref{prop:system-confluence}]
(1) By Lemma~\ref{lem:canonical-equivalent}, $R_r\subseteq\widehat R_r$.
Rows are never added, so a row that a \commitment event of $\rho_r$ removed is
not in $R_r$; hence $R_r\subseteq\widehat R_r\setminus\Delta(\rho_r)$.
Conversely, let $(k,u)\in\widehat R_r\setminus\Delta(\rho_r)$.  Both runs
start at the same canonical state, which contains $(k,u)$.  Consider an event of
$\rho_r$ at a state containing $(k,u)$.  If it is $\mathsf{input}(j)$, the
canonical run also takes $\mathsf{input}(j)$ and keeps $(k,u)$, so
$j\InputOrd k$ and the event keeps $(k,u)$.  If it is $\mathsf{output}(o)$,
then likewise $o\OutputOrd u$ and the event keeps $(k,u)$.  If it is a
\commitment event, it keeps $(k,u)$ because $(k,u)\notin\Delta(\rho_r)$.  By
induction on the events, $(k,u)\in R_r$.

(2) Suppose continuations bring every run to a common state $(k,R)$.  That
state has a row $(k,u)$, and events never add rows, so $(k,u)\in R_r$ for every
$r$.  Conversely, suppose $(k,u)\in R_r$ for every $r$.  Since
$i_r\InputOrd k$, the event $\mathsf{input}(k)$ is legal at $(i_r,R_r)$ and
keeps $(k,u)$, so $\mathsf{output}(u)$ is then enabled.  It ends at
$(k,R'_r)$ with $R'_r=R_r[\mathrm{in}\InputGe k,\ \mathrm{out}\OutputGe u]$.
If $R'_r$ has rows other than $(k,u)$, consider the partial function that maps
each state with current input $k$ whose table contains $(k,u)$ and some other
row, and whose rows at $k$ all \outrelbase $u$, to $(k,\{(k,u)\})$.  It keeps
the current input and removes a nonempty set of rows.  A removed row at $k$ is some $(k,x)$ with
$u\OutputOrd x$ and $x\neq u$, so $x\not\OutputOrd u$; a removed row at
another input has no remaining row at its input.  So the function is a
\commitment (Definition~\ref{def:commitment}), and a \commitment event that
applies it ends at $(k,\{(k,u)\})$.  If $R'_r$ has no other row, the run is
already there.  Every run therefore reaches $(k,\{(k,u)\})$.  By~(1),
$(k,u)\in R_r$ exactly when $(k,u)\in\widehat R_r$ and
$(k,u)\notin\Delta(\rho_r)$.
\end{proof}

\begin{proof}[Proof of Proposition~\ref{prop:commitment-conflict}]
Events only remove rows, so $(k,x)$ is present at $\sigma_P(i)$.  Thus $k$
\inrel $i$ and $x\in\fiber{k}$, so $\mathsf{input}(k)$ and then
$\mathsf{output}(x)$ are enabled from $\sigma_P(i)$, and a canonical second
run exists.  Under any basis, after $\mathsf{output}(x)$ every row at $k$
\outrel $x$.  The
\commitment leaves no such row at $k$
(Equation~\ref{eq:commitment-unsubsumed}); if it removed every row at $k$,
there is none at all.  Later events only remove rows, so no row at $k$
survives in both runs, and by Proposition~\ref{prop:system-confluence} the
runs are not confluent at $k$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:complete-cap}]
The canonical system for $E$ has the canonical basis and no \commitment
events.  Its strategies emit any output $o$ of $E$ in the current fiber, by
the canonical output event $\mathsf{output}(o)$ of
Section~\ref{sec:complete-calm}.  That event applies whatever \commitment $o$
requires, namely $x_o$ from the canonical basis
(Definition~\ref{def:run-under-basis}).  So the system has Availability for
$E$ whatever the spec.

Now suppose $\Spec$ satisfies Lifted Monotonicity for $E$.  Every output the
canonical system emits is the output of a row of $E$ whose input lies below
the current one.  So at a common \indesc $j$ of the current inputs, the
emitted outputs have a common \outnoun in $\fiber{j}$.  The system emits only
canonical output events, so Lemma~\ref{lem:reconciliation} makes this
confluence at $j$.  This gives Partition Consistency.

Conversely, suppose some system, such as the canonical system for $E$, has
Availability for $E$ and Partition Consistency.  Let
$(i_1,o_1),\ldots,(i_k,o_k)$ be rows of $E$ with a common \indesc $j$.  We
find some $u\in\fiber{j}$ that \outrel every $o_r$.  If $k=0$, then $j$
appears in some row $(j,u)$, since $\Spec$ is total.  Otherwise, take one
participant $p_r$ per row, starting at input $i_r$, each on its own side.  A
partitioned history has at least two sides, so if $k=1$ we add a second
participant for that row on another side.  Each participant runs without
messages.  By Availability for $E$, each $p_r$ can emit $o_r$.  By Partition
Consistency, the participants then reach one state $(j,R)$.  That state has
some row $(j,u)$ (Definition~\ref{def:residual-state}), and this $u$ is the
one we seek.  Rows are never added, so $(j,u)$ was already in each $p_r$'s
table just after it emitted $o_r$.  Emitting an output keeps only rows whose
output \outrel it (Definition~\ref{def:lifted-output-event}).  So $u$ \outrel
every $o_r$.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:complete-cap-any-system}]
The converse argument in the proof of Theorem~\ref{thm:complete-cap} applies
to any system with Availability for $E$ and Partition Consistency, so $\Spec$
satisfies Lifted Monotonicity for $E$.  By Theorem~\ref{thm:complete-cap},
$\Spec$ escapes CAP for $E$.
\end{proof}

\subsection{Invariant confluence (\S\ref{sec:iconfluence-instance})}
\label{app:iconfluence}

The input records which transactions ran and whose writes each one ran on.
A \emph{transaction history} $h$ (here simply \emph{history}) is a finite
partial order of transaction events, each labeled by a transaction in $T$.
Here $e'<e$ means that $e$ ran on a state that included the writes of $e'$.
A sub-history is \emph{closed} when it contains every event that precedes one
of its own events, that is, when it is a downward-closed sub-order.  History
$h'$ \inrel $h$ when $h$ is a closed sub-history of $h'$.  Then the events
that $h'$ adds precede no event of $h$, and each event of $h$ keeps exactly
the predecessors it had.

The state of a history is computed event by event.  An event $e$, labeled by
the transaction $t_e$, runs on
$\mathrm{pre}(e)=s_0\cup\bigcup_{e'<e}\mathrm{post}(e')$ and produces
$\mathrm{post}(e)=t_e(\mathrm{pre}(e))$.  The state of $h$ is
$\mathrm{eval}(h)=s_0\cup\bigcup_{e\in h}\mathrm{post}(e)$.  Because $h$ is a
partial order and transactions are inflationary ($t(s)\supseteq s$),
$\mathrm{pre}(e)$ is the state of the sub-history strictly below $e$, and
$\mathrm{post}(e)$ is the state of the sub-history at or below $e$.  A
closed sub-history keeps each event's predecessors, so it also keeps each
event's $\mathrm{pre}$ and $\mathrm{post}$.  Hence the state of a union of
closed sub-histories is the union of their states.

A history is \emph{locally valid} when every $\mathrm{pre}(e)$ and
$\mathrm{post}(e)$ satisfies $\mathcal I$.  This is the case when replicas
abort transactions that would violate the invariant on their own state.  A
history is \emph{valid} when the state of each of its closed sub-histories
satisfies $\mathcal I$, that is, $\mathcal I(\mathrm{eval}(c))$ for every
closed sub-history $c$.

The spec $\Spec_{\mathcal I}$ has the locally valid histories as inputs;
a closed sub-history of a locally valid history is locally valid, so the
inputs are closed under the order.
An output is a state, and $u$ \outrel $s$ when $s\subseteq u$.  At input $h$,
each valid closed sub-history of $h$ contributes its state as a permitted
output:
\[
  P_{\mathcal I}(h)=\{\mathrm{eval}(g):g\text{ a valid closed
  sub-history of }h\}.
\]
An output may therefore leave out some of the input's transactions, but what
it includes must be closed and valid.  Taking $g$ empty gives
$s_0$, so the spec is total.

\begin{proof}[Proof of Proposition~\ref{prop:iconfluence-lm}]
We use the common-extension form of Lifted Monotonicity
(Proposition~\ref{prop:common-expansion-form}).  First, the state of every
valid history $g$ is reachable.  The proof is by induction on $g$.  Let $e$ be
maximal in $g$.  Then $g\setminus\{e\}$ and the sub-history strictly below $e$
are closed, hence valid.  By induction their states
$\mathrm{eval}(g\setminus\{e\})$ and $\mathrm{pre}(e)$ are reachable.  The
state $\mathrm{post}(e)$ is a valid transaction step from $\mathrm{pre}(e)$,
and $\mathrm{eval}(g)=\mathrm{eval}(g\setminus\{e\})\cup\mathrm{post}(e)$ is a
valid merge.

($\Rightarrow$) Let rows $(h_r,\mathrm{eval}(g_r))$, $r=1,\ldots,k$, have a
common \indesc $j$.  Then each $g_r$ is a closed sub-history of $j$.  Let
$g=g_1\cup\cdots\cup g_k$; we show that $g$ is valid.  Every closed
sub-history $c$ of $g$ is the union of the pieces $c\cap g_r$.  Each piece is
a closed sub-history of the valid $g_r$, so its state is reachable by the
first step.  Thus $\mathrm{eval}(c)$ is a union of reachable states.
Invariant confluence makes each successive union valid, and a valid merge of
reachable states is reachable, so $\mathcal I(\mathrm{eval}(c))$.  Hence $g$
is valid, and $\mathrm{eval}(g)\in P_{\mathcal I}(j)$ \outrel every
$\mathrm{eval}(g_r)$.  For $k=0$, take $g$ empty.

($\Leftarrow$) Only the property of having directed fibers is needed.  We
first show that valid histories are closed under disjoint union.  Let $g_1$
and $g_2$ be valid histories.  Let $j$ be their disjoint union, with events
renamed apart and no order between the two parts.  Each event keeps its
predecessors, so its $\mathrm{pre}$ and $\mathrm{post}$ are states of closed
sub-histories of a valid $g_r$.  So $j$ is locally valid, and
$\mathrm{eval}(g_1),\mathrm{eval}(g_2)\in P_{\mathcal I}(j)$.  Because the
spec has directed fibers, there is a valid sub-history $g$ of $j$ with
$\mathrm{eval}(g)\supseteq\mathrm{eval}(g_1)\cup\mathrm{eval}(g_2)=\mathrm{eval}(j)\supseteq\mathrm{eval}(g)$.
So $\mathcal I(\mathrm{eval}(g_1)\cup\mathrm{eval}(g_2))$.  Every closed
$c\subseteq j$ is the disjoint union of $c\cap g_1$ and $c\cap g_2$, both
valid.  The same argument gives $\mathcal I(\mathrm{eval}(c))$, so $j$ is
valid.  Valid histories are also closed under adding an event above all
others when its transaction step is valid.  By induction on derivations, every
reachable state is therefore the state of a valid history.  The pairwise
argument then gives $\mathcal I(s_1\cup s_2)$ for all reachable $s_1,s_2$.
\end{proof}

\subsection{The HAT taxonomy (\S\ref{sec:hat-instance})}
\label{app:hat-taxonomy}

Bailis et al.\ classify transactional guarantees by what a system can provide
while every client that can reach a replica keeps getting
answers~\cite{bailis2013hat}.  The extended paper defines each guarantee by
the anomaly it rules out, a pattern of dependencies between transactions, and
gives algorithms for the guarantees that can be
provided~\cite{bailis2013hatextended}.  We use those definitions and
algorithms as written.  Where a definition disagrees with its own example, we
follow the example and list the differences at the end of this subsection.

\paragraph{Inputs and outputs.}
The inputs extend the histories of Definition~\ref{def:hat-spec}.  Each item
has a fixed set of replicas, and a read is served by a replica of its item.  A
\emph{predicate read} reads a fixed range of items at once and returns a
version of each.  A transaction's reads are ordered.  A client runs one
transaction at a time.  Its transactions form its \emph{session}, and its
commit timestamps increase along the session.  Outputs are sets of read
answers, as before.

\paragraph{Guarantees.}
Each guarantee rules out one anomaly of the extended paper: a pattern in a
graph whose nodes are transactions.  The edges are the write-, read-, and
anti-dependencies of Adya on each item~\cite{adya1999weakconsistency}, and
session edges from each transaction to its client's later ones.  For example,
read your writes rules out a transaction that reads a version of $x$ older
than one that an earlier transaction of the same client wrote.  The extended
paper gives no formal definition of recency.  We take it to require that a
read return a version at least as new as every write by another transaction
whose commit happens before the read.  A replica can return only versions it
holds.  So the \emph{visibility store} permits an answer only if its version is the initial
one or arrived at the reading replica before the read.

\begin{definition}[Guarantee spec, high availability, sticky read, sticky availability]
\label{def:hat-availability}
For a guarantee $G$, the table of $\Spec_G$ contains an output at an input when
the visibility store permits the output and the input extends to a complete
history that includes the output and has no anomaly that $G$ rules out.  The
guarantee $G$ is \emph{highly available} when some explicit relation
$E\subseteq P_G$ answers every read and some system for $\Spec_G$ has
Availability for $E$ and Partition Consistency.  A read is \emph{sticky} when
its replica has received every write of the client's earlier transactions to
the items that replica holds.  The guarantee $G$ is \emph{sticky available}
when the same holds for an $E$ that answers every sticky read.
\end{definition}

Sticky availability follows Bailis et al.: a client gets answers whenever its
transaction runs against a copy of the database that reflects all of its
earlier operations.  By Theorem~\ref{thm:complete-cap} and
Corollary~\ref{cor:complete-cap-any-system}, $G$ is highly available exactly
when $\Spec_G$ satisfies Lifted Monotonicity for some $E\subseteq P_G$ that
answers every read, and likewise for sticky availability.

\paragraph{Reveal rules.}
A transaction is \emph{pending stable} before a read when every replica of
every item it wrote has received its write before the read.  The algorithm of
Bailis et al.\ for monotonic atomic view reveals a write only once its
transaction is pending stable.  For the session guarantees they also wait
until every transaction committed before it is pending stable: they ``wait to
reveal new writes \ldots\ until each write's respective dependencies are
visible on all replicas.''  Their cut isolation algorithm keeps the versions a
transaction has read in a client-side cache.  We combine these into one
\emph{reveal rule}.  A write is revealed at a read once its transaction and
every transaction whose commit happens before its commit are pending stable.
A transaction reads, for each item, the highest version revealed at every one
of its reads so far.  The \emph{sticky reveal rule} answers only sticky reads
and also reveals the client's own earlier writes, which a sticky read's
replica holds.

\begin{theorem}[HAT taxonomy]
\label{thm:hat-taxonomy}
\leavevmode
\begin{enumerate}
  \item Under the timestamp condition of Theorem~\ref{thm:hat-levels}, read
        committed and monotonic atomic view are highly available.  Item cut
        isolation, predicate cut isolation, monotonic reads, monotonic
        writes, and writes follow reads are highly available.
  \item Read your writes, PRAM, and causal consistency are sticky available
        and not highly available.  Given at least two participants and any
        assignment of clients to them, consider the deployment that lets each
        client's participant give the sticky reveal rule's answers to that
        client's reads.  Some system has Deployment Availability and Partition
        Consistency for it.
  \item Recency is neither highly available nor sticky available.  For any
        level that rules out Lost Update or Write Skew, no system has
        Availability for an explicit relation that answers every read and
        Partition Consistency.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof sketch]
(1) The reveal rule answers every read.  Each answer depends only on the read's
past, so later input never changes it, and the rule is monotone.  A monotone
relation inside the spec satisfies Lifted Monotonicity for itself: rows below
$j$ have the relation's own answer at $j$ as a common \outnoun.  The rule
returns only versions the replica holds, and its answers have none of the
anomalies these guarantees rule out.  Consider monotonic reads.  An earlier
transaction of a client read some version of $x$, which was then revealed.
A revealed version has reached every replica of its item, so it stays
revealed at every later read at any replica.  The client's later transaction
reads after the earlier one commits, so it reads that version of $x$ or a newer
one.
For read committed and monotonic atomic view, the timestamp condition rules
out G1c, as in Theorem~\ref{thm:hat-levels}.  By
Theorem~\ref{thm:complete-cap}, each guarantee is highly available.

(2) The sticky reveal rule does the same for every sticky read.  The deployment
authorizes only answers of this one monotone relation inside the spec, so it
is monotone across partitions, and Proposition~\ref{prop:deployment-complete-cap}
gives the system.  Without stickiness, consider a client that writes $x$ at
one replica and then reads $x$ at another replica that has not received the
write.  That replica holds only the initial version of $x$, which read your
writes rules out.  No explicit relation inside the spec answers the read.
PRAM and causal consistency include read your writes.

(3) For recency, another client's write of $x$ commits before a read but never
reaches the reading replica, which again holds only the initial version.  The
read is sticky, so stickiness does not help.  For Write Skew, two transactions
at partitioned replicas each read the item the other writes.  Each replica can
return only the initial version, and together the answers form the cycle that
Write Skew names, as for Lost Update in Theorem~\ref{thm:hat-levels}.
\end{proof}

\paragraph{The algorithm for monotonic atomic view.}
The extended paper also gives pseudocode for monotonic atomic
view~\cite[App.~B]{bailis2013hatextended}.  A replica keeps a write in a
\emph{pending} set until it has received acknowledgments from every replica of
the items its transaction wrote, and then moves it to a \emph{good} set.  A
read of $x$ may carry a required timestamp.  Without one, the replica returns
its highest good version of $x$.  With one, it returns a good version at least
as new, or else the pending write with exactly that timestamp.  After each
read, the client raises the required timestamp of every item that the
returned write's transaction also wrote, and it clears these timestamps when
the transaction ends.  Which acknowledgments a replica has received is not
fixed by the input.  So we let each replica's good set be any set of pending
stable transactions that depends only on the read's past.  The pending case
returns only a write whose transaction an earlier read of the same transaction
found good, so every answer is a version the replica holds.

\begin{proposition}[The algorithm provides monotonic atomic view]
\label{prop:hat-mav-algorithm}
For every choice of the replicas' good sets, and under the timestamp condition
of Theorem~\ref{thm:hat-levels}, the answers of the algorithm for monotonic
atomic view form an explicit relation inside $P_{\mathrm{MAV}}$ that answers
every read and is monotone.  So monotonic atomic view is highly available.
\end{proposition}

\begin{proof}[Proof sketch]
Each answer depends only on the read's past, so the relation is monotone, and
it answers every read.  The timestamp condition rules out G1c, as for read
committed.  For the anomaly of monotonic atomic view, suppose a transaction
reads $x$ from $T_j$ and later reads $y$, which $T_j$ also wrote.  The first
read raised the required timestamp of $y$ to $T_j$'s, so the second read
returns $T_j$'s version of $y$ or a newer one.  Theorem~\ref{thm:complete-cap}
gives the system.
\end{proof}

\paragraph{Differences in form.}
Our classification matches theirs, and it differs in form in three ways.
First, our framework separates two reasons a guarantee can be unavailable.
The unavailable isolation levels fail Lifted Monotonicity: answers that the two
sides give have no common extension.  Read your writes without stickiness, and
recency in every case, fail earlier, because some read has no permitted answer
at all.  When a client's write has not reached the replica that serves its next
read, every version that replica holds, the initial version included, violates
read your writes.  Recency fails the same way when the missing write comes from
another client, so stickiness does not help.  No explicit relation inside the
spec answers such a read, so the question of Lifted Monotonicity never arises.
Second, monotonic writes needs the version order to follow each session's
order.  Bailis et al.\ obtain this from client timestamps, and our inputs
assume it: a client's commit timestamps increase along its session.  Third, we
model a predicate as a fixed range of items; general predicates, whose items
depend on the data, are not modeled.  Bailis et al.\ list several recency
guarantees without formal definitions, among them bounds on staleness and safe,
regular, and linearizable register semantics; we model the one recency
guarantee defined above.

\paragraph{Differences from the extended paper's text.}
Besides the definition of read your writes (\S\ref{sec:hat-instance}), the
extended paper's formal definitions differ from their own examples or prose in
the following places.  We follow the examples.
\begin{itemize}
  \item Def.~24 (predicate cut isolation) says that a history has the anomaly
        if, \emph{for all} pairs of overlapping predicate reads in a
        transaction, the transactions that change their matches differ.
        That holds trivially for a transaction with no such pair.  We read
        ``for some pair.''
  \item Def.~26 (monotonic atomic view) asks for a read-dependency on $x$ and
        ``a set of edges by $y$ containing at least one anti-dependency.''
        Read literally, this allows any path of edges on $y$, and the
        extended paper's own algorithm produces such a path.  In one history,
        $T_j$ writes $x$ and $y$.  A transaction $T_k$ reads $y$ before
        $T_j$'s writes are pending stable, so it gets the initial version, and
        then writes $y$ with a higher timestamp than $T_j$'s.  Then $T_i$
        reads $x$ and $y$, both from $T_j$.  The read-dependency from $T_j$ to
        $T_i$ on $x$ and the anti-dependencies from $T_i$ to $T_k$ and from
        $T_k$ to $T_j$ on $y$ form a cycle of the literal shape, although
        $T_i$ saw all of $T_j$'s writes.  The definition also says that
        ``$T_j$'s read from $y$ precedes its read from $x$,'' while its
        example (Fig.~9) has $T_i$ read $x$ before $y$.  We follow the
        example: $T_i$ reads $x$ from $T_j$ and later reads a version of $y$
        older than $T_j$'s.
  \item Def.~28 (monotonic reads) describes a cycle of exactly a session
        edge, an anti-dependency, and a read-dependency.  That misses a
        transaction that reads a version two or more versions older than one
        its client read earlier.  We allow write-dependencies between the
        anti-dependency and the read-dependency.
  \item Def.~30 (monotonic writes) requires a write-dependency edge, but its
        example (Figs.~13 and~14) has none.  We also rule out the example's
        anomaly.
  \item Def.~32 (writes follow reads) states the condition for the anomaly's
        absence (``$T_3$ does not directly anti-depend on $T_1$''), says
        ``write-depends'' where its example has read-dependencies, and names
        the anomaly MRWD, while Defs.~33 and~37 call it MWRD.  We read:
        $T_2$ reads $x$ from $T_1$, $T_3$ reads $y$ from $T_2$, and $T_3$
        reads the version of $x$ that $T_1$ overwrote.
  \item The legend of the dependency graph swaps the labels of read- and
        anti-dependencies relative to the figures, and Def.~14 reverses its
        indices.
\end{itemize}
